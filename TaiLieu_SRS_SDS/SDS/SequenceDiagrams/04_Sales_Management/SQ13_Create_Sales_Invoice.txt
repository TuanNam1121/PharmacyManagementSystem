actor "Pharmacist" as Pharmacist #7ACFF5
participant "<<boundary>>\nPosCounterPage" as PosCounterPage #7ACFF5

participant "<<boundary>>\nCustomerFormModal" as CustomerFormModal #7ACFF5
participant "<<boundary>>\nInvoicePrintModal" as InvoicePrintModal #7ACFF5
participant "<<control>>\nSalesInvoiceController" as InvoiceController #7ACFF5
participant "<<control>>\nSalesInvoiceServiceImpl" as InvoiceServiceImpl #7ACFF5
participant "<<entity>>\nCustomerRepository" as CustomerRepository #7ACFF5

database "<<entity>>\nSQL Server (PMS DB)" as DB #7ACFF5

activate Pharmacist #7ACFF5
Pharmacist -> PosCounterPage: Scan barcodes / search medicine, pick units (Tablet/Blister/Box)
activate PosCounterPage #7ACFF5

== Customer Identification (Mandatory) ==
opt Customer not yet identified (customerId == null)
    Pharmacist -> PosCounterPage: Click "+ Nhập SĐT"
    PosCounterPage -> CustomerLookupModal: Open modal
    activate CustomerLookupModal #7ACFF5
    
    Pharmacist -> CustomerLookupModal: Enter phone number
    CustomerLookupModal -> CustomerRepository: Search customer by phone
    activate CustomerRepository #7ACFF5
    CustomerRepository --> CustomerLookupModal: Return matching customer or Empty
    deactivate CustomerRepository
    
    alt Customer not found
        Pharmacist -> CustomerLookupModal: Click "+ Thêm Mới"
        CustomerLookupModal -> CustomerFormModal: Open modal on top
        activate CustomerFormModal #7ACFF5
        Pharmacist -> CustomerFormModal: Enter fullName, phone, allergy notes, click "Lưu khách hàng"
        CustomerFormModal --> PosCounterPage: Customer created, auto-select customer
        deactivate CustomerFormModal
    else Customer found
        Pharmacist -> CustomerLookupModal: Click "Chọn"
        CustomerLookupModal --> PosCounterPage: Set customerId & allergy info
    end
    deactivate CustomerLookupModal
end

PosCounterPage -> PosCounterPage: Check drug allergies against customer health profile
opt Customer has allergy to drug active ingredient
    PosCounterPage --> Pharmacist: Display red Alert: "Khách hàng dị ứng với hoạt chất [X]!"
end

== Finalize Order Submission ==
Pharmacist -> PosCounterPage: Click "Thanh toán (F9)"

PosCounterPage -> InvoiceController: POST /api/v1/sales-invoices (CreateSalesInvoiceRequest, Bearer AT)
activate InvoiceController #7ACFF5

InvoiceController -> InvoiceServiceImpl: createSalesInvoice(CreateSalesInvoiceRequest, accountId)
activate InvoiceServiceImpl #7ACFF5

InvoiceServiceImpl -> CustomerRepository: findById(request.getCustomerId())
activate CustomerRepository #7ACFF5
CustomerRepository -> DB: SELECT * FROM customers WHERE id=?
activate DB #7ACFF5
DB --> CustomerRepository: Customer entity
deactivate DB
CustomerRepository --> InvoiceServiceImpl: Customer
deactivate CustomerRepository

loop For each cart item (Deduct inventory - FEFO rule)
    InvoiceServiceImpl -> InventoryRepository: findNearestExpiringBatch(storeId, productId)
    activate InventoryRepository #7ACFF5
    InventoryRepository -> DB: SELECT * FROM inventories WHERE store_id=? AND product_id=? ORDER BY expiry_date ASC
    activate DB #7ACFF5
    DB --> InventoryRepository: Inventory batch
    deactivate DB
    InventoryRepository --> InvoiceServiceImpl: Inventory entity
    deactivate InventoryRepository
    
    InvoiceServiceImpl -> InventoryRepository: inv.setQuantity(inv.getQuantity() - baseQuantity)
    activate InventoryRepository #7ACFF5
    InventoryRepository -> DB: UPDATE inventories SET quantity=?, updated_at=NOW() WHERE id=?
    activate DB #7ACFF5
    DB --> InventoryRepository: OK
    deactivate DB
    InventoryRepository --> InvoiceServiceImpl: OK
    deactivate InventoryRepository
end

InvoiceServiceImpl -> InvoiceServiceImpl: generateInvoiceCode() -> 'INV-20261108-3910'
activate InvoiceServiceImpl #7ACFF5
deactivate InvoiceServiceImpl

InvoiceServiceImpl -> InvoiceRepository: save(SalesInvoice: code, finalAmount, shiftId, customerId)
activate InvoiceRepository #7ACFF5
InvoiceRepository -> DB: INSERT INTO sales_invoices (code, customer_id, shift_id, final_amount, status=1)
activate DB #7ACFF5
DB --> InvoiceRepository: SalesInvoice entity (id=3910)
deactivate DB
InvoiceRepository --> InvoiceServiceImpl: savedInvoice
deactivate InvoiceRepository

InvoiceServiceImpl -> InvoiceDetailRepo: saveAll(SalesInvoiceDetails)
activate InvoiceDetailRepo #7ACFF5
InvoiceDetailRepo -> DB: INSERT INTO sales_invoice_details (...)
activate DB #7ACFF5
DB --> InvoiceDetailRepo: OK
deactivate DB
InvoiceDetailRepo --> InvoiceServiceImpl: OK
deactivate InvoiceDetailRepo

InvoiceServiceImpl -> CustomerRepository: Add loyalty points (points + finalAmount/100000)
activate CustomerRepository #7ACFF5
CustomerRepository -> DB: UPDATE customers SET loyalty_points=? WHERE id=?
activate DB #7ACFF5
DB --> CustomerRepository: OK
deactivate DB
CustomerRepository --> InvoiceServiceImpl: OK
deactivate CustomerRepository

InvoiceServiceImpl --> InvoiceController: SalesInvoiceResponse
deactivate InvoiceServiceImpl

InvoiceController --> PosCounterPage: HTTP 201 Created + SalesInvoiceResponse
deactivate InvoiceController

PosCounterPage -> InvoicePrintModal: Auto-open K80 Receipt preview modal
activate InvoicePrintModal #7ACFF5
InvoicePrintModal --> Pharmacist: Display receipt ready for printing / customer handover
deactivate InvoicePrintModal

PosCounterPage --> Pharmacist: Clear shopping cart, ready for next order
deactivate PosCounterPage
deactivate Pharmacist
