@startuml SQ01_Login
title UC01 - Login Sequence Diagram
autonumber "1:"
skinparam style strictuml
skinparam sequenceMessageAlign left
skinparam defaultFontName Arial
skinparam defaultFontSize 12

skinparam sequence {
    LifeLineBorderColor #000000
    LifeLineBackgroundColor #7ACFF5
    ParticipantBorderColor #000000
    ParticipantBackgroundColor #7ACFF5
    ParticipantFontColor #000000
    ParticipantFontSize 12
    ActorBorderColor #000000
    ActorBackgroundColor #7ACFF5
    ActorFontColor #000000
    ArrowColor #000000
    ArrowFontColor #000000
    GroupBorderColor #000000
}

actor "User" as User #7ACFF5
participant "<<boundary>>\nLoginPage" as LoginPage #7ACFF5
participant "<<control>>\nAuthController" as AuthController #7ACFF5
participant "<<control>>\nAuthServiceImpl" as AuthServiceImpl #7ACFF5
participant "<<control>>\nBCryptPasswordEncoder" as BCryptEncoder #7ACFF5
participant "<<control>>\nJwtTokenProvider" as JwtProvider #7ACFF5
participant "<<entity>>\nAccountRepository" as AccountRepository #7ACFF5
participant "<<entity>>\nRefreshTokenRepository" as RefreshTokenRepo #7ACFF5
participant "<<entity>>\nAuditLogRepository" as AuditLogRepo #7ACFF5
database "<<entity>>\nSQL Server (PMS DB)" as DB #7ACFF5

activate User #7ACFF5
User -> LoginPage: Enter username/email and password, click "Login"
activate LoginPage #7ACFF5

LoginPage -> LoginPage: Client-side Validation (NotEmpty, trim)

LoginPage -> AuthController: POST /api/v1/auth/login (LoginRequest)
activate AuthController #7ACFF5

AuthController -> AuthServiceImpl: login(LoginRequest)
activate AuthServiceImpl #7ACFF5

AuthServiceImpl -> AccountRepository: findByUsernameOrEmail(username, username)
activate AccountRepository #7ACFF5

AccountRepository -> DB: SELECT a FROM accounts a WHERE a.username=? OR a.email=?
activate DB #7ACFF5
DB --> AccountRepository: Account record
deactivate DB

AccountRepository --> AuthServiceImpl: Optional<Account>
deactivate AccountRepository

alt Invalid Account or status != 1 (Locked)
    AuthServiceImpl --> AuthController: throw BadCredentialsException / AccountLockedException
    AuthController --> LoginPage: HTTP 401 Unauthorized / 403 Forbidden
    LoginPage --> User: Display error message ("Sai tài khoản hoặc đã bị khóa")
else Valid Account & Active Status
    AuthServiceImpl -> BCryptEncoder: matches(rawPassword, account.getPassword())
    activate BCryptEncoder #7ACFF5
    BCryptEncoder --> AuthServiceImpl: true (Password matches)
    deactivate BCryptEncoder

    AuthServiceImpl -> JwtProvider: generateAccessToken(account)
    activate JwtProvider #7ACFF5
    JwtProvider --> AuthServiceImpl: accessToken (JWT 15m)
    deactivate JwtProvider

    AuthServiceImpl -> JwtProvider: generateRefreshToken(account)
    activate JwtProvider #7ACFF5
    JwtProvider --> AuthServiceImpl: refreshToken (8h)
    deactivate JwtProvider

    AuthServiceImpl -> RefreshTokenRepo: save(RefreshToken entity)
    activate RefreshTokenRepo #7ACFF5
    RefreshTokenRepo -> DB: INSERT INTO refresh_tokens (token, account_id, expiry_date, is_revoked)
    activate DB #7ACFF5
    DB --> RefreshTokenRepo: OK
    deactivate DB
    RefreshTokenRepo --> AuthServiceImpl: savedRefreshToken
    deactivate RefreshTokenRepo

    AuthServiceImpl -> AuditLogRepo: save(AuditLog 'LOGIN_SUCCESS')
    activate AuditLogRepo #7ACFF5
    AuditLogRepo -> DB: INSERT INTO audit_logs (...)
    activate DB #7ACFF5
    DB --> AuditLogRepo: OK
    deactivate DB
    AuditLogRepo --> AuthServiceImpl: OK
    deactivate AuditLogRepo

    AuthServiceImpl --> AuthController: LoginResponse (accessToken, userProfile)
    deactivate AuthServiceImpl

    AuthController -> AuthController: Attach RefreshToken to HttpOnly Cookie (Set-Cookie)
    AuthController --> LoginPage: HTTP 200 OK + LoginResponse Body + Set-Cookie Header
    deactivate AuthController

    LoginPage -> LoginPage: Store accessToken in AuthContext Memory
    LoginPage --> User: Redirect to authorized Dashboard / POS Counter
end

deactivate LoginPage
deactivate User
@enduml
