@startuml SQ02_Logout
title UC02 - Logout Sequence Diagram
autonumber "1:"
skinparam style strictuml
skinparam sequenceMessageAlign left
skinparam defaultFontName Arial
skinparam defaultFontSize 12

skinparam sequence {
    LifeLineBorderColor #000000
    LifeLineBackgroundColor #7ACFF5
    ParticipantBorderColor #000000
    ParticipantBackgroundColor #7ACFF5
    ParticipantFontColor #000000
    ParticipantFontSize 12
    ActorBorderColor #000000
    ActorBackgroundColor #7ACFF5
    ActorFontColor #000000
    ArrowColor #000000
    ArrowFontColor #000000
    GroupBorderColor #000000
}

actor "User" as User #7ACFF5
participant "<<boundary>>\nMainLayout (Header)" as MainLayout #7ACFF5
participant "<<control>>\nAuthController" as AuthController #7ACFF5
participant "<<control>>\nAuthServiceImpl" as AuthServiceImpl #7ACFF5
participant "<<entity>>\nRefreshTokenRepository" as RefreshTokenRepo #7ACFF5
participant "<<entity>>\nAuditLogRepository" as AuditLogRepo #7ACFF5
database "<<entity>>\nSQL Server (PMS DB)" as DB #7ACFF5

activate User #7ACFF5
User -> MainLayout: Click "Logout" from Avatar Dropdown
activate MainLayout #7ACFF5

MainLayout -> AuthController: POST /api/v1/auth/logout (Bearer accessToken, Cookie: refreshToken)
activate AuthController #7ACFF5

AuthController -> AuthServiceImpl: logout(refreshToken, accountId, response)
activate AuthServiceImpl #7ACFF5

opt refreshToken exists
    AuthServiceImpl -> RefreshTokenRepo: findByToken(refreshToken)
    activate RefreshTokenRepo #7ACFF5
    RefreshTokenRepo -> DB: SELECT * FROM refresh_tokens WHERE token=?
    activate DB #7ACFF5
    DB --> RefreshTokenRepo: RefreshToken entity
    deactivate DB
    RefreshTokenRepo --> AuthServiceImpl: RefreshToken
    deactivate RefreshTokenRepo

    AuthServiceImpl -> RefreshTokenRepo: revoke(token)
    activate RefreshTokenRepo #7ACFF5
    RefreshTokenRepo -> DB: UPDATE refresh_tokens SET is_revoked=1 WHERE token=?
    activate DB #7ACFF5
    DB --> RefreshTokenRepo: OK
    deactivate DB
    RefreshTokenRepo --> AuthServiceImpl: OK
    deactivate RefreshTokenRepo
end

AuthServiceImpl -> AuditLogRepo: save(AuditLog 'LOGOUT_SUCCESS')
activate AuditLogRepo #7ACFF5
AuditLogRepo -> DB: INSERT INTO audit_logs (...)
activate DB #7ACFF5
DB --> AuditLogRepo: OK
deactivate DB
AuditLogRepo --> AuthServiceImpl: OK
deactivate AuditLogRepo

AuthServiceImpl -> AuthController: Set-Cookie: refreshToken=; Max-Age=0; Path=/api/v1/auth
deactivate AuthServiceImpl

AuthController --> MainLayout: HTTP 200 OK (Logged out successfully)
deactivate AuthController

MainLayout -> MainLayout: Clear accessToken in Memory, reset Auth State
MainLayout --> User: Redirect to Login Page (/login)

deactivate MainLayout
deactivate User
@enduml
